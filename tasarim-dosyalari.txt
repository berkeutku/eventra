import type { Metadata } from "next";
import Link from "next/link";
import CategoryChips from "@/components/CategoryChips";
import EventCard from "@/components/EventCard";
import EventFilters from "@/components/EventFilters";
import { categoryMeta, type EventCategory } from "@/data/events";
import {
  countByCategory,
  listCities,
  listEvents,
  parseWhen,
  realEventsEnabled,
  type SourceFilter,
} from "@/lib/events";

export const dynamic = "force-dynamic";

export const metadata: Metadata = {
  title: "Etkinlikler | Eventra",
  description: "Yaklaşan etkinlikleri keşfet ve ortak yolculuk planla.",
};

type Param = string | string[] | undefined;

type Props = {
  searchParams: Promise<Record<string, Param>>;
};

function first(value: Param): string {
  return (Array.isArray(value) ? value[0] : value) ?? "";
}

export default async function EventsPage({ searchParams }: Props) {
  const params = await searchParams;
  const realEnabled = realEventsEnabled();

  const category = (Object.keys(categoryMeta) as EventCategory[]).find(
    (item) => item === first(params.kategori),
  );
  const cityRaw = first(params.sehir);
  const cityKey = /^[a-z0-9-]{1,40}$/.test(cityRaw) ? cityRaw : "";
  const query = first(params.q).trim().slice(0, 80);
  const sourceRaw = first(params.kaynak);
  const source: SourceFilter =
    realEnabled && (sourceRaw === "real" || sourceRaw === "demo")
      ? sourceRaw
      : "all";
  const when = parseWhen(first(params.tarih));
  const pageRaw = Number.parseInt(first(params.sayfa), 10);
  const page = Number.isFinite(pageRaw) && pageRaw > 0 ? pageRaw : 1;

  const filters = {
    category,
    cityKey: cityKey || undefined,
    query: query || undefined,
    source,
    when,
  };

  const [result, cities, counts] = await Promise.all([
    listEvents({ ...filters, page }),
    listCities(filters),
    countByCategory(filters),
  ]);

  function pageHref(target: number): string {
    const search = new URLSearchParams();

    if (category) {
      search.set("kategori", category);
    }
    if (cityKey) {
      search.set("sehir", cityKey);
    }
    if (query) {
      search.set("q", query);
    }
    if (source !== "all") {
      search.set("kaynak", source);
    }
    if (when !== "all") {
      search.set("tarih", when);
    }
    if (target > 1) {
      search.set("sayfa", String(target));
    }

    const text = search.toString();
    return text ? `/events?${text}` : "/events";
  }

  const navButton =
    "rounded-full border border-line bg-white px-5 py-2.5 text-sm font-bold text-ink transition-colors hover:bg-lilac";

  return (
    <main className="px-6 py-10 sm:py-20">
      <div className="mx-auto max-w-6xl">
        <h1 className="text-3xl font-extrabold tracking-tight sm:text-6xl">
          Bugün hangi etkinliğe gidiyoruz?
        </h1>
        <p className="mt-4 max-w-xl text-lg text-muted">
          Katılacağın etkinliği seç, aynı yöne gidenlerle yolculuk planla.
        </p>

        <EventFilters
          query={query}
          cityKey={cityKey}
          cities={cities}
          category={category}
          source={source}
          when={when}
          showSource={realEnabled}
        />

        <div className="mt-6">
          <CategoryChips
            active={category}
            counts={counts}
            params={{
              sehir: cityKey || undefined,
              q: query || undefined,
              kaynak: source !== "all" ? source : undefined,
              tarih: when !== "all" ? when : undefined,
            }}
          />
        </div>

        <p className="mt-4 text-sm text-muted">
          {result.total} etkinlik ·{" "}
          {realEnabled
            ? "Gerçek etkinlik bilgileri biletimGO'dan alınır ve değişebilir. Demo etkinlik etiketli olanlar kurgusaldır."
            : "Bu sayfadaki etkinlikler demo amaçlı kurgusal verilerdir."}
        </p>

        {result.items.length === 0 ? (
          <div className="mt-10 rounded-3xl bg-white p-10 text-center ring-1 ring-black/5">
            <p className="font-bold text-ink">Bu filtrelere uyan etkinlik yok</p>
            <p className="mt-1 text-sm text-muted">
              Filtreleri değiştirebilir ya da tüm etkinlikleri görebilirsin.
            </p>
            <Link
              href="/events"
              className="mt-4 inline-block text-sm font-bold text-violet hover:underline"
            >
              Tümünü gör
            </Link>
          </div>
        ) : (
          <div className="mt-8 grid gap-6 sm:grid-cols-2 lg:grid-cols-3">
            {result.items.map((event) => (
              <EventCard key={event.id} event={event} headingLevel={2} />
            ))}
          </div>
        )}

        {result.pageCount > 1 ? (
          <nav
            aria-label="Sayfalama"
            className="mt-10 flex flex-wrap items-center justify-center gap-4"
          >
            {result.page > 1 ? (
              <Link href={pageHref(result.page - 1)} className={navButton}>
                Önceki
              </Link>
            ) : null}
            <span className="text-sm font-bold text-ink">
              Sayfa {result.page} / {result.pageCount}
            </span>
            {result.page < result.pageCount ? (
              <Link href={pageHref(result.page + 1)} className={navButton}>
                Sonraki
              </Link>
            ) : null}
          </nav>
        ) : null}
      </div>
    </main>
  );
}

"use client";

import Link from "next/link";
import { useState, useTransition } from "react";
import { useToast } from "@/components/Toast";
import { toggleFavoriteAction } from "@/lib/favorite-actions";

const base =
  "inline-flex items-center justify-center gap-2 rounded-full border px-5 py-2.5 text-sm font-bold transition-colors disabled:cursor-not-allowed disabled:opacity-60";

export default function FavoriteButton({
  eventId,
  initial,
  loggedIn,
}: {
  eventId: string;
  initial: boolean;
  loggedIn: boolean;
}) {
  const toast = useToast();
  const [favorite, setFavorite] = useState(initial);
  const [pending, startTransition] = useTransition();

  if (!loggedIn) {
    return (
      <Link
        href={`/giris?next=/events/${eventId}`}
        className={`${base} border-line text-ink hover:bg-lilac`}
      >
        <span aria-hidden="true">♡</span>
        Favorilere eklemek için giriş yap
      </Link>
    );
  }

  function handleClick() {
    const want = !favorite;

    startTransition(async () => {
      const result = await toggleFavoriteAction(eventId, want);

      if (result.ok) {
        setFavorite(result.favorited);
        toast.show(
          result.favorited ? "Favorilere eklendi." : "Favorilerden çıkarıldı.",
        );
      } else {
        toast.show(result.message, "info");
      }
    });
  }

  return (
    <button
      type="button"
      onClick={handleClick}
      disabled={pending}
      aria-pressed={favorite}
      className={`${base} ${
        favorite
          ? "border-violet bg-lilac text-violet"
          : "border-line text-ink hover:bg-lilac"
      }`}
    >
      <span aria-hidden="true">{favorite ? "♥" : "♡"}</span>
      {favorite ? "Favorilerde" : "Favorilere ekle"}
    </button>
  );
}

  const [busy, setBusy] = useState(false);

  useEffect(() => {
    let cancelled = false;

    async function detect() {
      let next: PushState;

      if (process.env.NODE_ENV !== "production") {
        next = "dev";
      } else if (!PUBLIC_KEY) {
        next = "no-key";
      } else if (!isPushSupported()) {
        next = isIos() && !isStandalone() ? "needs-install" : "unsupported";
      } else if (Notification.permission === "denied") {
        next = "denied";
      } else {
        const registration = await navigator.serviceWorker.getRegistration();
        const subscription = await registration?.pushManager.getSubscription();

        if (subscription) {
          // Cihazdaki abonelik bu hesaba bağlı olsun (başka hesapla girildiyse eşitler).
          await saveToServer(subscription).catch(() => false);
          next = "on";
        } else {
          next = "off";
        }
      }

      if (!cancelled) {
        setState(next);
      }
    }

    detect().catch(() => {
      if (!cancelled) {
        setState("unsupported");

        setState(permission === "denied" ? "denied" : "off");

        toast.show("Bildirim servisi hazır değil. Sayfayı yenileyip dene.", "info");

        (await registration.pushManager.getSubscription()) ??

          userVisibleOnly: true,
          applicationServerKey: urlBase64ToUint8Array(PUBLIC_KEY),

        await subscription.unsubscribe();
        toast.show("Bildirimler kaydedilemedi. Lütfen tekrar dene.", "info");
        return;

      setState("on");
      toast.show("Bu cihazda telefon bildirimleri açıldı.");

export default function RideList({
  rides,
  eventId,
  gate,
}: {
  rides: RideItem[];
  eventId: string;
  gate: TicketGate;
}) {
  const [query, setQuery] = useState("");
  const [minSeats, setMinSeats] = useState("0");
  const [sort, setSort] = useState<SortKey | null>(null);
  const [destination, setDestination] = useState("");
  const [preferredTime, setPreferredTime] = useState("");

  const scored = useMemo(
    () => scoreRides(rides, { destination, preferredTime }),
    [rides, destination, preferredTime],
  );

  if (rides.length === 0) {
    return <EmptyRides eventId={eventId} gate={gate} />;
  }

  const hasPreference = destination.trim() !== "" || preferredTime !== "";
  const activeSort: SortKey = sort ?? (hasPreference ? "match" : "time");
  const showScore = activeSort === "match";
  const normalizedQuery = normalize(query.trim());
  const minSeatCount = Number(minSeats);

  const visibleRides = scored
    .filter((item) => matchesQuery(item.ride, normalizedQuery))
    .filter((item) => remainingSeats(item.ride) >= minSeatCount)
    .sort((a, b) => {
      if (activeSort === "match") {
        if (a.full !== b.full) {
          return a.full ? 1 : -1;
        }
        if (b.score !== a.score) {
          return b.score - a.score;
        }
        return (
          departureSortKey(a.ride.departureTime) -
          departureSortKey(b.ride.departureTime)
        );
      }

      if (activeSort === "cost") {
        return (
          calcSharePerPerson(a.ride.totalCost, a.ride.seats) -
          calcSharePerPerson(b.ride.totalCost, b.ride.seats)
        );
      }

      return (
        departureSortKey(a.ride.departureTime) -
        departureSortKey(b.ride.departureTime)
      );
    });

  const bestId =
    showScore && hasPreference
      ? visibleRides.find((item) => !item.full && item.score > 0)?.ride.id
      : undefined;

  return (

  currentRole,
  onDark = false,
}: {
  currentRole: Role;
  onDark?: boolean;
}) {
  const toast = useToast();
  const [pending, startTransition] = useTransition();

  function choose(role: Role) {
    if (role === currentRole || pending) {
      return;
    }

    const data = new FormData();
    data.set("role", role);

    startTransition(async () => {
      const ok = await switchRoleAction(data);

      if (ok) {
        toast.show(`${roleLabels[role]} rolüne geçtin.`);
      } else {
        toast.show("Rol değiştirilemedi. Lütfen tekrar dene.", "info");
      }
    });
  }

  return (
    <div
      role="group"
      aria-label="Aktif rol"
      className={`inline-flex rounded-full p-1 ${onDark ? "bg-white/15" : "bg-lilac"}`}
    >
      {ORDER.map((role) => {
        const active = role === currentRole;
        const on = onDark ? "bg-white text-violet" : "bg-violet text-white";
        const off = onDark
          ? "text-white hover:bg-white/10"
          : "text-violet hover:bg-white";

        return (
          <button
            key={role}
            type="button"
            onClick={() => choose(role)}
            aria-pressed={active}
            disabled={pending}
            className={`min-h-10 rounded-full px-5 text-sm font-bold transition-colors disabled:cursor-wait ${active ? on : off}`}

import { useToast } from "@/components/Toast";

export default function ShareButton({ title }: { title: string }) {
  const toast = useToast();

  async function handleShare() {
    const url = window.location.href;

    try {
      if (typeof navigator.share === "function") {
        await navigator.share({ title, url });
        return;
      }

      await navigator.clipboard.writeText(url);
      toast.show("Bağlantı panoya kopyalandı.");
    } catch (error) {
      // Kullanıcı paylaşım penceresini kapattıysa hata sayılmaz.
      if (error instanceof DOMException && error.name === "AbortError") {
        return;
      }

      toast.show(
        "Bağlantı paylaşılamadı. Adres çubuğundan kopyalayabilirsin.",
        "info",
      );
    }
  }

  return (
    <button
      type="button"
      onClick={handleShare}
      className="inline-flex items-center justify-center rounded-full border border-line px-5 py-2.5 text-sm font-bold text-ink transition-colors hover:bg-lilac"
    >
      Etkinliği paylaş
    </button>
  );
}

"use client";

import {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useMemo,
  useRef,
  useState,
  type ReactNode,
} from "react";
import Icon from "@/components/Icons";

type Tone = "success" | "info";

type ToastItem = {
  id: number;
  message: string;
  tone: Tone;
};

type ToastApi = {
  show: (message: string, tone?: Tone) => void;
};

const ToastContext = createContext<ToastApi | null>(null);

export function useToast(): ToastApi {
  const context = useContext(ToastContext);

  if (!context) {
    throw new Error("useToast yalnızca ToastProvider içinde kullanılabilir.");
  }

  return context;
}

export default function ToastProvider({ children }: { children: ReactNode }) {
  const [items, setItems] = useState<ToastItem[]>([]);
  const nextId = useRef(1);
  const timers = useRef(new Map<number, ReturnType<typeof setTimeout>>());

  const dismiss = useCallback((id: number) => {
    clearTimeout(timers.current.get(id));
    timers.current.delete(id);
    setItems((current) => current.filter((item) => item.id !== id));
  }, []);

  const show = useCallback(
    (message: string, tone: Tone = "success") => {
      const id = nextId.current;
      nextId.current += 1;

      // Ekranda en fazla 3 bildirim kalır.
      setItems((current) => [...current.slice(-2), { id, message, tone }]);
      timers.current.set(
        id,
        setTimeout(() => dismiss(id), 5000),
      );
    },
    [dismiss],
  );

  useEffect(() => {
    const activeTimers = timers.current;

    return () => {
      activeTimers.forEach((timer) => clearTimeout(timer));
      activeTimers.clear();
    };
  }, []);

  const api = useMemo(() => ({ show }), [show]);

  return (
    <ToastContext.Provider value={api}>
      {children}
      <div
        role="status"
        aria-live="polite"
        className="pointer-events-none fixed inset-x-0 bottom-24 z-50 flex flex-col items-center gap-2 px-4 sm:bottom-6"
      >
        {items.map((item) => (
          <div
            key={item.id}
            className="pointer-events-auto flex max-w-md items-center gap-3 rounded-full bg-night py-2.5 pl-3 pr-2 text-sm font-bold text-white shadow-xl shadow-black/20"
          >
            <span
              className={`grid h-7 w-7 shrink-0 place-items-center rounded-full ${
                item.tone === "success"
                  ? "bg-mint text-night"
                  : "bg-white/20 text-white"
              }`}
            >
              <Icon name="check" className="h-4 w-4" />
            </span>
            <span>{item.message}</span>
            <button
              type="button"
              onClick={() => dismiss(item.id)}
              aria-label="Bildirimi kapat"
              className="grid h-8 w-8 shrink-0 place-items-center rounded-full text-white/70 hover:bg-white/10 hover:text-white"
            >
              <span aria-hidden="true">×</span>
            </button>
          </div>
        ))}
      </div>
    </ToastContext.Provider>
  );
}

import { cleanText } from "./rating-rules";

// Saf kurallar: veritabanı içermez, hem sunucuda hem tarayıcıda ve betiklerde kullanılabilir.

export const MAX_DESTINATION = 80;
export const MAX_NOTE = 140;
export const MAX_REQUESTS_PER_DAY = 5;
export const MAX_REQUEST_SEATS = 4;

const TIME_PATTERN = /^([01]\d|2[0-3]):[0-5]\d$/;
const WHOLE_NUMBER = /^\d+$/;

export type RequestField = "destination" | "preferredTime" | "seats" | "note";

export type RequestFields = {
  destination: string;
  preferredTime: string | null;
  seats: number;
  note: string | null;
};

export type RequestParse =
  | { ok: true; value: RequestFields }
  | { ok: false; errors: Partial<Record<RequestField, string>> };

// Telefon numarası, e-posta ya da bağlantı gibi iletişim bilgisi içeriyor mu?
export function looksLikeContact(text: string): boolean {
  return (
    /(?:\d[\s\-.()]*){10,}/.test(text) ||
    text.includes("@") ||
    /https?:|www\./i.test(text)
  );
}

export function parseRequestInput(
  input: Record<RequestField, string>,
): RequestParse {
  const errors: Partial<Record<RequestField, string>> = {};

  const destination = cleanText(input.destination, MAX_DESTINATION);
  const note = cleanText(input.note, MAX_NOTE);
  const timeText = input.preferredTime.trim();
  const seatsText = input.seats.trim();

  if (destination.length < 2) {
    errors.destination = "Varış yeri en az 2 karakter olmalı.";
  } else if (looksLikeContact(destination)) {
    errors.destination =
      "Varış yerine telefon, e-posta ya da bağlantı yazma, yalnızca yer adı yaz.";
  }

  if (timeText !== "" && !TIME_PATTERN.test(timeText)) {
    errors.preferredTime = "Saat geçerli değil.";
  }

  const seats = WHOLE_NUMBER.test(seatsText) ? Number(seatsText) : null;

  if (seats === null || seats < 1 || seats > MAX_REQUEST_SEATS) {
    errors.seats = `Kişi sayısı 1 ile ${MAX_REQUEST_SEATS} arasında olmalı.`;
  }

  if (note && looksLikeContact(note)) {
    errors.note =
      "Nota telefon, e-posta ya da bağlantı yazma. İletişim yolculuk oluşturulduktan sonra kurulur.";
  }

  if (Object.keys(errors).length > 0 || seats === null) {
    return { ok: false, errors };
  }

  return {
    ok: true,
    value: {
      destination,
      preferredTime: timeText === "" ? null : timeText,
      seats,
      note: note === "" ? null : note,
    },
  };
}

// Yalnızca ilk ad gösterilir (gizlilik).
export function firstName(name: string): string {
  const first = name.trim().split(/\s+/)[0] ?? "";
  return first ? first.slice(0, 20) : "Yolcu";
}

import { and, desc, eq, inArray, lt, sql } from "drizzle-orm";
import { db } from "../db";

  rideMessages,
  rides,
  ticketVerifications,

  MAX_MESSAGES_PER_RIDE,
  MAX_MESSAGES_PER_WINDOW,
  MESSAGE_WINDOW_MS,
  NOTIFY_BUCKET_MS,
  RETENTION_DAYS,
  chatPhase,
  prepareMessage,
  type ChatItem,
  type ChatPhase,
} from "./chat-rules";
import { takeRateSlot } from "./rate-limit";
import { firstName } from "./request-rules";

// Bu dosya hem uygulamadan hem test betiğinden çağrıldığı için göreli içe aktarımlar kullanıyor.

const UUID_PATTERN =
  /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;

export type ChatAccess = {
  rideId: string;
  eventId: string;
  // Yalnızca bu etkinlik için bileti doğrulanmış katılımcılar.
  participantIds: string[];
  phase: ChatPhase;
};

export type ChatStatus =
  | { status: "ok"; access: ChatAccess }
  | { status: "verify_required"; eventId: string }
  | { status: "unavailable" }
  | { status: "none" };

// Sohbet için iki şart gerekir: kişi yolculuğun sürücüsü ya da o an katılmış bir yolcusu olmalı
// VE bu etkinliğin biletini doğrulamış olmalı. Gerçek (biletimGO) etkinliklerde sohbet kapalıdır.
export async function getChatStatus(
  rideId: string,
  userId: string,
  now: Date = new Date(),
): Promise<ChatStatus> {
  if (!UUID_PATTERN.test(rideId)) {
    return { status: "none" };
  }

  const [row] = await db
    .select({
      eventId: rides.eventId,
      driverId: rides.driverId,
      source: events.source,
      startsAt: events.startsAt,
      endsAt: events.endsAt,
    })
    .from(rides)
    .innerJoin(events, eq(rides.eventId, events.id))
    .where(eq(rides.id, rideId))
    .limit(1);

  if (!row) {
    return { status: "none" };
  }

  const passengers = await db
    .select({ userId: ridePassengers.userId })
    .from(ridePassengers)
    .where(eq(ridePassengers.rideId, rideId));

  const everyone = [row.driverId, ...passengers.map((p) => p.userId)];

  if (!everyone.includes(userId)) {
    return { status: "none" };
  }

  if (row.source === "biletimgo") {
    return { status: "unavailable" };
  }

  const verifiedRows = await db
    .select({ userId: ticketVerifications.userId })
    .from(ticketVerifications)
    .where(
      and(
        eq(ticketVerifications.eventId, row.eventId),
        inArray(ticketVerifications.userId, everyone),
      ),
    );

  const verified = new Set(verifiedRows.map((item) => item.userId));

  if (!verified.has(userId)) {
    return { status: "verify_required", eventId: row.eventId };
  }

  return {
    status: "ok",
    access: {
      rideId,
      eventId: row.eventId,
      participantIds: everyone.filter((id) => verified.has(id)),
      phase: chatPhase(
        {
          startsAt: row.startsAt.toISOString(),
          endsAt: row.endsAt ? row.endsAt.toISOString() : null,
        },
        now,
      ),
    },
  };
}

export async function getChatAccess(
  rideId: string,
  userId: string,
  now: Date = new Date(),
): Promise<ChatAccess | null> {
  const result = await getChatStatus(rideId, userId, now);
  return result.status === "ok" ? result.access : null;
}

// Çağıran taraf erişimi önceden doğrulamış olmalı. Son 60 mesaj, eskiden yeniye.
export async function listMessages(
  rideId: string,
  viewerId: string,
  limit = 60,
): Promise<ChatItem[]> {
  const rows = await db
    .select({
      id: rideMessages.id,
      senderId: rideMessages.senderId,
      name: users.name,
      body: rideMessages.body,
      createdAt: rideMessages.createdAt,
    })
    .from(rideMessages)
    .innerJoin(users, eq(rideMessages.senderId, users.id))
    .where(eq(rideMessages.rideId, rideId))
    .orderBy(desc(rideMessages.createdAt), desc(rideMessages.id))
    .limit(limit);

  return rows.reverse().map((row) => ({
    id: row.id,
    mine: row.senderId === viewerId,
    name: firstName(row.name),
    body: row.body,
    at: row.createdAt.toISOString(),
  }));
}

export async function loadChat(
  rideId: string,
  userId: string,
  now: Date = new Date(),
): Promise<{ access: ChatAccess; items: ChatItem[] } | null> {
  const access = await getChatAccess(rideId, userId, now);

  if (!access) {
    return null;
  }

  return { access, items: await listMessages(rideId, userId) };
}

export type SendStatus =
  | "sent"
  | "empty"
  | "too_long"
  | "link"
  | "not_participant"
  | "read_only"
  | "rate_limited"
  | "ride_full";

export async function sendMessage(input: {
  rideId: string;
  userId: string;
  text: unknown;
  now?: Date;
}): Promise<SendStatus> {
  const now = input.now ?? new Date();
  const prepared = prepareMessage(input.text);

  if (!prepared.ok) {
    return prepared.reason;
  }

  const access = await getChatAccess(input.rideId, input.userId, now);

  if (!access) {
    return "not_participant";
  }

  if (access.phase === "readonly") {
    return "read_only";
  }

  const allowed = await takeRateSlot(
    input.userId,
    "chat_message",
    MAX_MESSAGES_PER_WINDOW,
    MESSAGE_WINDOW_MS,
    now,
  );

  if (!allowed) {
    return "rate_limited";
  }

  const [counted] = await db
    .select({ total: sql<number>`count(*)::int` })
    .from(rideMessages)
    .where(eq(rideMessages.rideId, input.rideId));

  if (counted.total >= MAX_MESSAGES_PER_RIDE) {
    return "ride_full";
  }

  await db.insert(rideMessages).values({
    rideId: input.rideId,
    senderId: input.userId,
    body: prepared.text,
    createdAt: now,
  });

  // Diğer (doğrulanmış) katılımcılara bildirim: mesaj içeriği bildirimde yazmaz, aynı yolculuk için 5 dakikada bir.
  try {
    const recipients = access.participantIds.filter((id) => id !== input.userId);

    if (recipients.length > 0) {
      const [sender] = await db
        .select({ name: users.name })
        .from(users)
        .where(eq(users.id, input.userId))
        .limit(1);
      const bucket = Math.floor(now.getTime() / NOTIFY_BUCKET_MS);

      await db
        .insert(notifications)
        .values(
          recipients.map((userId) => ({
            userId,
            type: "ride_message" as const,
            title: "Yolculuğunda yeni mesaj var",
            body: `${firstName(sender?.name ?? "")} yolculuk sohbetine mesaj yazdı.`,
            link: `/events/${access.eventId}/yolculuk/${input.rideId}`,
            dedupeKey: `ride_message:${input.rideId}:${bucket}`,
          })),
        )
        .onConflictDoNothing();
    }
  } catch (error) {
    console.error(
      "Sohbet bildirimi yazılamadı:",
      error instanceof Error ? error.message : "bilinmeyen hata",
    );
  }

  // 60 günden eski mesajlar temizlenir. Hata gönderimi etkilemez.
  try {
    await db
      .delete(rideMessages)
      .where(
        lt(
          rideMessages.createdAt,
          new Date(now.getTime() - RETENTION_DAYS * 86_400_000),
        ),
      );
  } catch (error) {
    console.error(
      "Eski mesajlar silinemedi:",
      error instanceof Error ? error.message : "bilinmeyen hata",
    );
  }

  return "sent";
}

export async function deleteMessage(
  userId: string,
  messageId: string,
): Promise<boolean> {
  if (!UUID_PATTERN.test(messageId)) {
    return false;
  }

  const removed = await db
    .delete(rideMessages)
    .where(
      and(eq(rideMessages.id, messageId), eq(rideMessages.senderId, userId)),

    .returning({ id: rideMessages.id });

}

// Her yolculuk satırı için o yolculuğa katılan yolcu sayısını hesaplar.
function passengerCountSql() {
  return sql<number>`(select count(*)::int from ${ridePassengers} where ${ridePassengers.rideId} = ${rides.id})`;
}

// Sürücü bu etkinlik için (demo) bilet doğrulaması yapmış mı?
function driverVerifiedSql() {
  return sql<boolean>`exists (select 1 from ${ticketVerifications} where ${ticketVerifications.userId} = ${rides.driverId} and ${ticketVerifications.eventId} = ${rides.eventId})`;
}

// Sürücünün tüm yolculuklardan aldığı ortalama puan (tek ondalık) ve değerlendirme sayısı.
function driverRatingSql() {
  return sql<number | null>`(select round(avg(${rideRatings.score})::numeric, 1)::float8 from ${rideRatings} where ${rideRatings.rateeId} = ${rides.driverId})`;
}

function driverRatingCountSql() {
  return sql<number>`(select count(*)::int from ${rideRatings} where ${rideRatings.rateeId} = ${rides.driverId})`;
}

function toRideItem(
  ride: RideRow,
  driverName: string,
  passengerCount: number,
  driverVerified: boolean,
  flags: { joined: boolean; isOwn: boolean },
  rating?: { value: number | null; count: number },
): RideItem {
  return {
    id: ride.id,
    eventId: ride.eventId,
    driverName,
    origin: ride.origin,
    destination: ride.destination,
    meetingPoint: ride.meetingPoint,
    // Veritabanı "23:30:00" verir, arayüzde "23:30" gösteriyoruz.
    departureTime: ride.departureTime.slice(0, 5),
    durationMinutes: ride.durationMinutes,
    seats: ride.seats,
    totalCost: ride.totalCost,
    createdAt: ride.createdAt.toISOString(),
    passengerCount,
    joined: flags.joined,
    isOwn: flags.isOwn,
    driverVerified,
    driverRating: rating ? rating.value : null,
    driverRatingCount: rating ? rating.count : 0,
  };
}

export async function getRidesForEvent(
  eventId: string,
  userId?: string,
): Promise<RideItem[]> {
  const rows = await db
    .select({
      ride: rides,
      driverName: users.name,
      passengerCount: passengerCountSql(),
      driverVerified: driverVerifiedSql(),
      driverRating: driverRatingSql(),
      driverRatingCount: driverRatingCountSql(),
    })
    .from(rides)
    .innerJoin(users, eq(rides.driverId, users.id))
    .where(eq(rides.eventId, eventId))
    .orderBy(desc(rides.createdAt));

  const joinedIds = new Set<string>();

  if (userId && rows.length > 0) {
    const joined = await db
      .select({ rideId: ridePassengers.rideId })
      .from(ridePassengers)
      .where(
        and(
          eq(ridePassengers.userId, userId),
          inArray(
            ridePassengers.rideId,
            rows.map((row) => row.ride.id),
          ),
        ),
      );

    for (const item of joined) {
      joinedIds.add(item.rideId);
    }
  }

  return rows.map((row) =>
    toRideItem(
      row.ride,
      row.driverName,
      row.passengerCount,
      row.driverVerified,
      {
        joined: joinedIds.has(row.ride.id),
        isOwn: userId !== undefined && row.ride.driverId === userId,
      },
      { value: row.driverRating, count: row.driverRatingCount },
    ),
  );
}

export async function getRideDetail(
  rideId: string,
  userId?: string,
): Promise<RideDetail | null> {
  const [row] = await db
    .select({
      ride: rides,
      driverName: users.name,
      passengerCount: passengerCountSql(),
      driverVerified: driverVerifiedSql(),
      driverRating: driverRatingSql(),
      driverRatingCount: driverRatingCountSql(),
    })
    .from(rides)
    .innerJoin(users, eq(rides.driverId, users.id))
    .where(eq(rides.id, rideId))
    .limit(1);

  if (!row) {
    return null;
  }

  const passengerRows = await db
    .select({
      userId: ridePassengers.userId,
      name: users.name,
      verified: sql<boolean>`exists (select 1 from ${ticketVerifications} where ${ticketVerifications.userId} = ${ridePassengers.userId} and ${ticketVerifications.eventId} = ${row.ride.eventId})`,
    })
    .from(ridePassengers)
    .innerJoin(users, eq(ridePassengers.userId, users.id))
    .where(eq(ridePassengers.rideId, rideId))
    .orderBy(asc(ridePassengers.joinedAt));

  const isOwn = userId !== undefined && row.ride.driverId === userId;
  const joined =
    userId !== undefined &&
    passengerRows.some((passenger) => passenger.userId === userId);

  return {
    ...toRideItem(
      row.ride,
      row.driverName,
      row.passengerCount,
      row.driverVerified,
      { joined, isOwn },
      { value: row.driverRating, count: row.driverRatingCount },
    ),
    passengers:
      isOwn || joined
        ? passengerRows.map((passenger) => ({
            name: passenger.name,
            verified: passenger.verified,
          }))
        : null,
  };
}

function isUniqueViolation(error: unknown): boolean {
  const direct = (error as { code?: string } | null)?.code;
  const nested = (error as { cause?: { code?: string } } | null)?.cause?.code;
  return direct === "23505" || nested === "23505";
}

export async function generateDemoCodeAction(): Promise<GenerateCodeResult> {
  const user = await getCurrentUser();

  if (!user) {
    return { message: "Önce giriş yapmalısın." };
  }

  return { code: generateDemoTicketCode() };

export async function verifyTicketAction(
  _previous: VerifyTicketState,
  formData: FormData,
): Promise<VerifyTicketState> {
  const user = await getCurrentUser();

  if (!user) {
    return { message: "Önce giriş yapmalısın." };
  }

  const eventId = String(formData.get("eventId") ?? "");
  const code = String(formData.get("code") ?? "").trim();

  if (code.length === 0) {
    return { message: "Lütfen bilet kodunu gir." };
  }

  if (code.length > 40) {
    return { message: "Kod çok uzun. Örnek biçim: EVT-7KQ-M2X" };
  }

  const event = await getEventById(eventId);

  if (!event) {
    return { message: "Etkinlik bulunamadı." };
  }

  if (await isTicketVerified(user.id, event.id)) {
    return { ok: true };
  }

  const check = await checkTicket(code);

  if (check.status === "malformed") {
    return { message: "Kod biçimi hatalı. Örnek biçim: EVT-7KQ-M2X" };
  }

  if (check.status === "rejected") {
    return {
      message:
        "Simüle edilen sağlayıcı bu kodu reddetti (sağlama karakteri tutmuyor). Yazım hatası olabilir.",
    };
  }

  const codeHash = createHash("sha256").update(check.normalized).digest("hex");

  try {
    await db
      .insert(ticketVerifications)
      .values({ userId: user.id, eventId: event.id, codeHash });
  } catch (error) {
    if (isUniqueViolation(error)) {
      // Aynı anda iki kez gönderildiyse doğrulama zaten kaydedilmiştir.
      if (await isTicketVerified(user.id, event.id)) {
        return { ok: true };
      }

      return { message: "Bu bilet kodu başka bir hesapta zaten kullanılmış." };
    }

    console.error("Bilet doğrulaması kaydedilemedi:", error);
    return {
      message: "Doğrulama kaydedilemedi. Lütfen biraz sonra tekrar dene.",
    };
  }

  revalidatePath("/events/[id]", "page");
  revalidatePath("/panel");
  return { ok: true };
}
